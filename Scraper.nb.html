# Helper function to detect date formats for a list of strings
detect_date_formats <- function(date_strings) {
  # A list of candidate formats (date-only and datetime variants)
  candidate_formats <- c(
    "%Y-%m-%dT%H:%M:%S",   # ISO 8601 with T separator
    "%Y-%m-%d %H:%M:%S",   # Common datetime with space separator
    "%Y-%m-%d",            # ISO date
    "%d/%m/%Y %H:%M:%S",   # Day/Month/Year with datetime
    "%d/%m/%Y",            # Day/Month/Year
    "%m/%d/%Y %H:%M:%S",   # Month/Day/Year with datetime
    "%m/%d/%Y",            # Month/Day/Year
    "%Y/%m/%d %H:%M:%S",   # Year/Month/Day with datetime using slash
    "%Y/%m/%d",            # Year/Month/Day with slash
    
    # Full month name formats like "September 23, 2020"
    "%B %d, %Y",
    "%B %d %Y",
    "%B %d, %Y %H:%M:%S",

    # Abbreviated month name formats like "Sep 23, 2021" (without a period)
    "%b %d, %Y",
    "%b %d, %Y %H:%M:%S",
    
    # Abbreviated month name with a period: "Sep. 23, 2021"
    "%b. %d, %Y",
    "%b. %d, %Y %H:%M:%S"
  )
  
  # Function to clean ordinal suffixes (st, nd, rd, th) from date strings
  clean_ordinal_suffixes <- function(date_str) {
    gsub("(\\d{1,2})(st|nd|rd|th)", "\\1", date_str, perl = TRUE)
  }
  
  # For each string in date_strings, find the first format (if any) that parses successfully
  sapply(date_strings, function(ds) {
    fmt_found <- NA_character_
    
    # Remove ordinal suffixes before checking formats
    cleaned_ds <- clean_ordinal_suffixes(ds)
    
    for (fmt in candidate_formats) {
      parsed <- suppressWarnings(strptime(cleaned_ds, format = fmt))
      if (!is.na(parsed)) {
        fmt_found <- fmt
        break
      }
    }
    
    fmt_found
  })
}

# Loops through known nodes for elements (titles, dates, PR links and text)
node_hunter <- function(link, debug=F){
  
  # Check if link has https:// at the start, if not add it
  if(substr(link, 1, 8)!="https://") link <- paste0("https://", link)
  
  # Check if link is in valid format and date range is valid
  if(!grepl("https://[a-zA-Z0-9\\-]+\\.house\\.gov", link)){
    stop("Link is not of the form \"{name}.house.gov\" or \"https://{name}.house.gov\"")
  }
  
  # Sanitize input by removing extra forward slash at end of URL if present
  if(substr(link, nchar(link), nchar(link))=="/") link <- substr(link, 1, nchar(link)-1)
  
  # Initialize variables we'll return
  pr_page_link <- NA
  page_string <- NA
  page_index_start <- NA
  date_node <- NA
  date_node_type <- NA
  date_formats <- NA
  title_node <- NA
  title_node_type <- NA
  pr_link_node <- NA
  pr_link_node_type <- NA
  text_node <- NA
  text_node_type <- NA
  edgecase <- F # Indicator for non-paginated cases
  
  # Read in our webpage elements to look through
  nodes <- read_excel("nodes.xlsx")
  
  # Catch edge cases that require different scraping approach
  if(grepl("https://crane.house.gov", link)){
    
    edgecase <- T
    
    pr_page_link <- paste0(link, "/media")
    
    pr_page <- try(read_html(pr_page_link), silent=T)
    
    if(!inherits(pr_page, "try-error") & length(pr_page)==2){
      pr_page_link_home <- pr_page_link
    }
    else{
      stop("Edge case detected but failed to find PR page for: ", link)
    }
  }
  # Otherwise figure out link for the PR page and pagination
  else{
    
    # Find the link for the press release page
    pr_page_link <- presser_page_finder(link, debug=debug)
    
    if(is.na(link)) stop(paste0("Press release page could not be found for ", link))
    
    # Build the URL for the press releases page
    if (!grepl("^https?://", pr_page_link)) {
      # Remove trailing slashes from link
      base_link <- sub("/+$", "", link)
      
      # Remove leading slashes from pr_page_link
      relative_link <- sub("^/+", "", pr_page_link)
      
      # Combine with a single slash between them
      pr_page_link <- paste0(base_link, "/", relative_link)
    }
    
    # Save PR page home to return later
    pr_page_link_home <- pr_page_link
    
    ## FINDING PAGINATION FORMAT
    
    pagination <- nodes %>% filter(element=="page link")
    
    page_index_start <- NA
    page_string <- NA
    
    # Loop through press release page link formats until one works, then determine
    # if the pagination starts at 0 or 1
    for(p in pagination$string){
      page0 <- try(read_html(paste0(pr_page_link, p, 0)), silent=T)
      page1 <- try(read_html(paste0(pr_page_link, p, 1)), silent=T)
      page2 <- try(read_html(paste0(pr_page_link, p, 2)), silent=T)
      
      # A non-blank page will have length > 1
      if(length(page0)==1 & length(page0)==1) next
      else if(identical(page0 %>% html_text(), page1 %>% html_text) &
              identical(page1 %>% html_text(), page2 %>% html_text)){ next }
      else if(length(page0) > length(page1)){ page_index_start <- 0; page_string <- p; break }
      else if(length(page0) < length(page1)){ page_index_start <- 1; page_string <- p; break }
      else if(length(page0)==2 & length(page1)==2 & identical(page0 %>% html_text(), 
                                                              page1 %>% html_text())){
        page_index_start <- 1; page_string <- p; break 
      }
      else if(length(page0)==2 & length(page1)==2 & !identical(page0 %>% html_text(), 
                                                               page1 %>% html_text())){
        if(page0 %>% html_text() %>% str_detect("There are no records to display")){
          page_index_start <- 1; page_string <- p; break
        }
        else{ page_index_start <- 0; page_string <- p; break }
      }
      else stop("Determining starting page index failed for: ", link)
    }
    
    if(is.na(page_string)) stop(paste0("Pagination failed for ", link))
    
    # Build the press release page URL
    pr_page_link <- paste0(pr_page_link, page_string, page_index_start)
    
    if(debug) message("URL: ", pr_page_link)
  }
  
  ## FINDING DATE NODE
  
  date_nodes <- nodes %>% filter(element=="dates")
  date_node <- NA
  date_node_type <- NA
  date_formats <- NA
  date_attempt <- character(0) # Not NA b/c we differentiate successes by length
                               # Might have a page with 1 presser
  
  # Loop through date nodes until one works
  for(i in 1:nrow(date_nodes)){
    # Case when date can be found with xpath
    if(date_nodes$type[i]=="xpath"){
      date_attempt <- try(read_html(pr_page_link) %>% 
                          html_elements(xpath=date_nodes$string[i]) %>% 
                            html_text() %>% str_trim(), silent=T)
    }
    # Case when date can be found with CSS
    else{
      date_attempt <- try(read_html(pr_page_link) %>% 
                            html_elements(date_nodes$string[i]) %>% 
                            html_text() %>% str_trim(), silent=T)
    }
    
    if(length(date_attempt)==0 | inherits(date_attempt, "try-error")) next
    else{
      
      # Format the dates, also remove any likely non-date junk in our vector
      # The longest normally formatted date (i.e. "September 31st, 2025") is 20
      # characters long
      date_attempt <- gsub("  ", " ", date_attempt)
      date_attempt <- gsub("(\\d{1,2})(st|nd|rd|th)", "\\1", date_attempt, perl = TRUE)
      date_attempt <- date_attempt[nchar(date_attempt)<=20]
      
      if(length(date_attempt)==0) next
      
      # Detect the format of each scraped date
      date_formats <- detect_date_formats(date_attempt)
      
      # If they're all NA then try next node
      if(all(is.na(date_formats))) next
      # Otherwise we have our date node
      else{
        date_attempt <- date_attempt[which(!is.na(date_formats))] # Remove non-date elements
        date_node <- date_nodes$string[i]
        date_node_type <- date_nodes$type[i] 
        break
      }
    }
  }
  
  if(all(is.na(date_formats))) stop(paste0("No valid date node found for ", link))
  
  if(debug) message("Date node: ", date_node, 
                    "\nDate format(s) detected: ", paste(unique(date_formats), collapse=" | "))
  
  ## FINDING TITLE LINK NODE
  
  title_nodes <- nodes %>% filter(element=="titles")
  title_node <- NA
  title_node_type <- NA
  title_attempt <- character(0)
  title_attempt_original_length <- NA
  
   # Loop through title nodes until one works
  for(i in 1:nrow(title_nodes)){
    
    # Case when title can be found with xpath
    if(title_nodes$type[i]=="xpath"){
      title_attempt <- try(read_html(pr_page_link) %>% 
                          html_elements(xpath=title_nodes$string[i]) %>% 
                            html_text() %>% str_trim(), silent=T)
    }
    # Case when title can be found with CSS
    else{
      title_attempt <- try(read_html(pr_page_link) %>% 
                            html_elements(title_nodes$string[i]) %>% 
                             html_text() %>% str_trim(), silent=T)
    }

    if(length(title_attempt)<1 | inherits(title_attempt, "try-error") |
       length(title_attempt) < length(date_attempt)) next 
    else{
      title_attempt_original_length <- length(title_attempt)
      
      # If title_attempt has more elements than date_attempt, grab the last elements
      # corresponding to the number of dates--sometimes headers are scraped before titles
      if(length(title_attempt) > length(date_attempt)){
        #title_attempt <- title_attempt[(length(title_attempt) - length(date_attempt) + 1):length(title_attempt)]
        next
      }
     
      title_node <- title_nodes$string[i]
      title_node_type <- title_nodes$type[i]
      break
    }
  }
  
  if(is.na(title_node)) stop(paste0("No valid title node found for ", link))
  
  if(debug) message("Title node: ", title_node)
  
  ## FINDING PR LINK NODE
  
  pr_link_attempt <- NA
  
  # Usually press release links are hyperlinked in the title so try that first
  pr_link_attempt <- try(read_html(pr_page_link) %>% 
                            html_elements(title_node) %>% html_attr("href") %>% 
                            str_trim(), silent=T)

  # If grabbing using the title node didn't work then loop through known
  # PR link nodes
  if(all(is.na(pr_link_attempt)) | inherits(pr_link_attempt, "try-error") |
     length(pr_link_attempt) > title_attempt_original_length){
    pr_link_nodes <- nodes %>% filter(element=="links")
    pr_link_node <- NA
    pr_link_attempt <- character(0)
    
    # Loop through PR link nodes until one works
    for(i in 1:nrow(pr_link_nodes)){
      
      # Case when date can be found with xpath
      if(pr_link_nodes$type[i]=="xpath"){
        pr_link_attempt <- try(read_html(pr_page_link) %>% 
                            html_elements(xpath=pr_link_nodes$string[i]) %>% 
                              html_text() %>% str_trim(), silent=T)
      }
      # Case when PR link can be found with CSS
      else{
        pr_link_attempt <- try(read_html(pr_page_link) %>% 
                               html_elements(pr_link_nodes$string[i]) %>% 
                               html_attr("href") %>% str_trim(), silent=T)
      }
  
      if(length(pr_link_attempt)<1 | inherits(pr_link_attempt, "try-error")) next 
      
      if(length(pr_link_attempt)<1 | inherits(pr_link_attempt, "try-error") |
       length(pr_link_attempt) < length(date_attempt)) next 
      else{
        # If pr_link_attempt has more elements than date_attempt, grab the last elements
        # corresponding to the number of dates--sometimes headers are scraped before pr_links
        if(length(pr_link_attempt) > length(date_attempt)){
          pr_link_attempt <- pr_link_attempt[(length(pr_link_attempt) - length(date_attempt) + 1):length(pr_link_attempt)]
        }
        pr_link_node <- pr_link_nodes$string[i]
        pr_link_node_type <- pr_link_nodes$type[i]
        break
      }
    }
    
  }
  # Otherwise PR link node and type are same as title
  else{
    # If pr_link_attempt has more elements than date_attempt, grab the last elements
    # corresponding to the number of dates--sometimes headers are scraped before pr_links
    if(length(pr_link_attempt) > length(date_attempt)){
      pr_link_attempt <- pr_link_attempt[(length(pr_link_attempt) - length(date_attempt) + 1):length(pr_link_attempt)]
    }
    
    pr_link_node <- title_node
    pr_link_node_type <- title_node_type
  }
  
  if(debug) message("PR link node:", pr_link_node)
  
  # Check if each PR link is a path or a complete URL and format accordingly
  pr_link_attempt1 <- format_pr_link(pr_link_attempt, pr_page_link)
  
  try_pr_link <- try(read_html(pr_link_attempt1[1]), silent=T)
  
  # If the first attempt fails it's probably because the individual press release
  # URLs don't include the path to the press releases page
  if(inherits(try_pr_link, "try-error")){
    pr_link_attempt2 <- format_pr_link(pr_link_attempt, link)

    # If that doesn't work, throw an error
    if(inherits(try(read_html(pr_link_attempt2[1]), silent=T), "try-error")){
      stop("Failed to build individual press release links for: ", link)
    }
    # Otherwise we now have our pr_link_attempt
    else pr_link_attempt <- pr_link_attempt2
  }
  else pr_link_attempt <- pr_link_attempt1
  
  if(debug) message("Example PR link: ", pr_link_attempt[1])
  
  
  # Check to make sure we have the same number of successfully parsed dates,
  # titles, and PR links
  # If not, then throw an error because something went wrong
  if(length(unique(c(sum(is.na(date_attempt)),  # Cheeky way to test equality of
                     sum(is.na(title_attempt)), # multiple values
                     sum(is.na(pr_link_attempt)))
                   ))!=1){
    if(debug){
      message("Dates: ", paste(date_attempt, collapse=" | "), "\n",
              "Titles:", paste(title_attempt, collapse=" | "), "\n",
              "Links: ", paste(pr_link_attempt, collapse=" | "))
    }
    stop("Number of dates, titles, and/or PR links does not match.")
  }
  
  ## FINDING PR TEXT NODE
  
  text_nodes <- nodes %>% filter(element=="text")
  text_link <- pr_link_attempt[1]
  text_node <- NA
  text_node_type <- NA
  text_attempt <- NA
  
  # Loop through text nodes until one works
  for(i in 1:nrow(text_nodes)){
    
    # Case when text can be found with xpath
    if(text_nodes$type[i]=="xpath"){
      text_attempt <- try(read_html(text_link) %>% 
                          html_element(xpath=text_nodes$string[i]) %>% 
                            html_text() %>% str_trim(), silent=T)
    }
    # Case when text can be found with CSS
    else{
      text_attempt <- try(read_html(text_link) %>% 
                            html_element(text_nodes$string[i]) %>% 
                             html_text() %>% str_trim(), silent=T)
    }
    
    if(is.na(text_attempt) | length(text_attempt)<1 | 
       inherits(text_attempt, "try-error")) next 
    else{
      text_node <- text_nodes$string[i]
      text_node_type <- text_nodes$type[i]
      break
    }
  }
  
  if(is.na(text_node)) stop(paste0("No valid PR text node found for ", link))
  
  if(debug) message("Text node: ", text_node)
  
  return(list(homepage_link=link,
              pr_page_link=pr_page_link_home,
              page_string=page_string,
              page_index_start=page_index_start,
              date_node=date_node,
              date_node_type=date_node_type,
              date_formats=date_formats,
              title_node=title_node,
              title_node_type=title_node_type,
              pr_link_node=pr_link_node,
              pr_link_node_type=pr_link_node_type,
              text_node=text_node,
              text_node_type=text_node_type,
              edgecase=edgecase # TRUE when legislator requires non-paginated scraping
              ))
  
}

edgecase_scraper <- function(info, start_date, end_date, 
                             pressers=data.frame(date=c(), title=c(), body=c(), 
                                                 tags=c(), url=c()),
                             clean_text=F, debug=F, progress=T){
  
  # Store scraping info in individual variables to make life easier
  homepage_link <- info$homepage_link
  pr_page_link <- info$pr_page_link
  date_node <- info$date_node
  date_node_type <- info$date_node_type
  date_formats <- info$date_formats
  title_node <- info$title_node
  title_node_type <- info$title_node_type
  pr_link_node <- info$pr_link_node
  pr_link_node_type <- info$pr_link_node_type
  text_node <- info$text_node
  text_node_type <- info$text_node_type
  
  start_date <- as.Date(start_date)
  end_date <- as.Date(end_date)
  
  # Get all the dates in the range
  date_range <- seq(from = start_date, to = end_date, by = "day")
  
  # Format the dates to be month year string we can use to navigate
  # press releases by month for legislators whose sites don't paginate
  monthyears <- unique(format(date_range, "/%Y/%m"))
  
  current_page <- NA
  
  for(m in monthyears){
    
    if(progress) message(ifelse(debug, "", "\014"), "Scraping pressers from: ", homepage_link, 
                       "    ", "/Year/Month: ", m)
    
    current_page <- try(read_html(paste0(homepage_link, m)), silent=T)
    
    if(inherits(current_page, "try-error")) next
    
    if(date_node_type=="css"){
      dates <- try(current_page %>% html_elements(date_node) %>% 
                            html_text() %>% str_trim, silent=T)
    }
    else{
      dates <- try(current_page %>% html_elements(xpath=date_node) %>% 
                            html_text() %>% str_trim, silent=T)
    }

  # Data frame that will hold scraped pressers
  pressers <- data.frame(date=c(), title=c(), body=c(), tags=c(), url=c())

  if(page_index>page_limit){
    message("Page limit reached for: ", homepage_link)
    return(pressers)
  }
  
  page_index <- ifelse(page_index==0, page_index_start, page_index)
  
  if(progress) message(ifelse(debug, "", "\014"), "Scraping pressers from: ", homepage_link, 
                       "    ", "pg: ", page_index)

  # Handle empty page case, i.e. when all the pages have been checked
    if(length(dates)==0) return(pressers)

      return(pressers)
